**Goal:** 집중 위젯 1개 제한, '크게' 모드 우측 패널의 편집 가능한 미니 스티키, 설정 헤더 대비 강화, 바로가기 6칸을 구현한다.

**Architecture:** 기존 PySide6/qfluentwidgets 데스크톱 앱. UI 로직은 `ui_*.py`, 데이터는 `database.py`(`_connection()` 컨텍스트 매니저 + `PRAGMA user_version` 마이그레이션), 보안 암호화는 `security.py`. 신규 미니 스티키는 별도 모듈 `ui_sticky_panel.py`로 분리하고 `ui_main.py`가 스케일 전환 시 이를 띄운다.

**Tech Stack:** Python 3.13, PySide6, qfluentwidgets, SQLite, pytest + pytest-qt.

```python
        from ui_focus import FocusWidgetWindow
        try:
            widgets = db.get_open_focus_widgets()
            factor = self._get_app_opacity()
            for idx, w in enumerate(widgets):
                if idx >= 1:
                    # 1개 제한: 초과분은 데이터 보존하되 닫힘 처리(창 미표시)

                win.setWindowOpacity(factor)
                QTimer.singleShot(0, lambda _w=win: (_w.show(), _w.raise_()))
                self._focus_windows[w["id"]] = win
        except (sqlite3.Error, OSError) as e:
            _logger.error("집중 모드 위젯 로드 실패: %s", e)
```

```python
def test_migration_v15_preserves_existing_and_idempotent(app_dir):
    """v14 상태(1~5만 존재)에서 migrate 시 기존 데이터 보존 + 멱등."""
    with db._connection() as conn:
        for i in range(1, 6):
            conn.execute(
                "UPDATE shortcuts SET name=?, path=? WHERE id=?",
                (f"name{i}", f"C:/app{i}.exe", i),
            )
        conn.execute("DELETE FROM shortcuts WHERE id = 6")
        conn.execute("PRAGMA user_version = 14")
        conn.commit()

    db.migrate()

    rows = {r["id"]: r for r in db.get_shortcuts()}
    assert len(rows) == 6
    assert rows[1]["name"] == "name1"
    assert rows[5]["path"] == "C:/app5.exe"
    assert rows[6]["name"] == ""
    assert rows[6]["path"] == ""

    # 멱등성: 다시 실행해도 6개 유지
    db.migrate()
    assert len(db.get_shortcuts()) == 6
```

```python
# ui_sticky_panel.py - '크게' 모드 우측 컬럼의 편집 가능한 미니 스티키 패널
import logging

from PySide6.QtCore import Qt, QTimer
from PySide6.QtGui import QColor
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QScrollArea

import database as db
from ui_sticky import _TodoTextEdit, STICKY_HEADER_COLORS

_logger = logging.getLogger("app_error")


class StickyMiniCard(QWidget):
    """우측 패널용 편집 가능한 미니 스티키 카드 (노트 1개 바인딩)."""

    def __init__(self, note_id: int, content: str, color: str, parent=None):
        super().__init__(parent)
        self.note_id = note_id
        self._color = color
        self.setObjectName("stickyMiniCard")
        self.setAttribute(Qt.WA_StyledBackground, True)

        layout = QVBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(0)

        # 상단 색상 스트립 (헤더 색 매핑 재사용)
        header_hex = STICKY_HEADER_COLORS.get(color, color)
        self._strip = QWidget(self)
        self._strip.setObjectName("miniStrip")
        self._strip.setFixedHeight(8)
        self._strip.setAttribute(Qt.WA_StyledBackground, True)
        self._strip.setStyleSheet(
            f"#miniStrip{{background:{header_hex};"
            "border-top-left-radius:8px;border-top-right-radius:8px;}}"
        )
        layout.addWidget(self._strip)

        # 본문 편집 (체크박스 토글 동작 상속)
        self.text_edit = _TodoTextEdit(self)
        if content.strip().startswith("<"):
            self.text_edit.setHtml(content)
        else:
            self.text_edit.setPlainText(content)
        self.text_edit.setStyleSheet(
            "QTextEdit{background:transparent;border:none;font-size:13px;padding:6px;}"
        )
        self.text_edit.setMinimumHeight(90)
        layout.addWidget(self.text_edit)

        # 카드 배경/테두리 (노트 배경색 기준)
        c = QColor(color)
        r, g, b = c.red(), c.green(), c.blue()
        br, bg, bb = max(0, r - 30), max(0, g - 30), max(0, b - 30)
        self.setStyleSheet(
            f"#stickyMiniCard{{background:rgba({r},{g},{b},255);"
            f"border:1px solid rgba({br},{bg},{bb},180);border-radius:8px;}}"
        )

        # 자동 저장 (2초 디바운스)
        self._save_timer = QTimer(self)
        self._save_timer.setSingleShot(True)
        self._save_timer.timeout.connect(self._auto_save)
        self.text_edit.textChanged.connect(lambda: self._save_timer.start(2000))

    def _auto_save(self):
        try:
            db.update_sticky_note(self.note_id, content=self.text_edit.toHtml())
        except (OSError, ValueError) as e:
            _logger.error("미니 스티키 저장 실패 (id=%s): %s", self.note_id, e)

    def save(self):
        """디바운스 무시 즉시 저장 (외부 호출용)."""
        self._save_timer.stop()
        self._auto_save()


class StickySidePanel(QWidget):
    """'크게' 모드 우측 컬럼의 스티키 노트 미니 패널."""

    def __init__(self, parent=None):
        super().__init__(parent)
        self.setObjectName("stickySidePanel")
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setStyleSheet(
            "#stickySidePanel{background:#FFFFFF;"
            "border:1px solid rgba(180,190,220,180);border-radius:12px;}"
        )
        self._cards = []

        outer = QVBoxLayout(self)
        outer.setContentsMargins(12, 12, 12, 12)
        outer.setSpacing(8)

        title = QLabel("스티키 노트", self)
        title.setStyleSheet("font-size:13px;font-weight:700;color:#212121;")
        outer.addWidget(title)

        self._scroll = QScrollArea(self)
        self._scroll.setWidgetResizable(True)
        self._scroll.setFrameShape(QScrollArea.NoFrame)
        self._scroll.setStyleSheet("QScrollArea{background:transparent;border:none;}")
        outer.addWidget(self._scroll, 1)

        self._list = QWidget()
        self._list.setStyleSheet("background:transparent;")
        self._list_layout = QVBoxLayout(self._list)
        self._list_layout.setContentsMargins(0, 0, 0, 0)
        self._list_layout.setSpacing(10)
        self._list_layout.addStretch()
        self._scroll.setWidget(self._list)

        # 빈 상태 라벨 (stretch 앞에 배치)
        self._empty_label = QLabel("표시할 스티키 노트가 없습니다.", self._list)
        self._empty_label.setStyleSheet("color:#888888;font-size:12px;")
        self._list_layout.insertWidget(0, self._empty_label)

    def reload(self):
        """열린 스티키 노트로 카드 목록을 재구성한다."""
        for card in self._cards:
            self._list_layout.removeWidget(card)
            card.deleteLater()
        self._cards = []

        try:
            notes = db.get_open_sticky_notes()
        except (OSError, ValueError) as e:
            _logger.error("미니 스티키 목록 조회 실패: %s", e)
            notes = []

        for note in notes:
            card = StickyMiniCard(
                note["id"], note["content"], note["color"], self._list
            )
            self._cards.append(card)
            # stretch 바로 앞(마지막에서 한 칸 앞)에 삽입
            self._list_layout.insertWidget(self._list_layout.count() - 1, card)

        self._empty_label.setVisible(not self._cards)

    def save_all(self):
        """모든 카드 즉시 저장."""
        for card in self._cards:
            card.save()
```
